const USERS=[

["cr.im3","CRIM3@10Minute","CR Manager — IM3","Customer Relations","CI"],
["cr.3id","CR3ID@10Minute","CR Manager — 3ID","Customer Relations","C3"],
["reporting.manager","Report@10Minute","Reporting Manager","Reporting","RP"],
["support.manager","Support@10Minute","Support / Other Manager","Support","SU"],["admin.feri","Feri@10Minute","Feri Susanto — Head of Ops / Administrator","Administrator","FS"]
];

const iso=()=>{let d=new Date();return new Date(d.getTime()-d.getTimezoneOffset()*60000).toISOString().slice(0,10)};
const key=(d,u)=>"10MU_V51_"+d+"_"+u;
const get=(d,u)=>{try{return JSON.parse(localStorage.getItem(key(d,u))||"null")}catch(e){return null}};
const put=(d,u,x)=>localStorage.setItem(key(d,u),JSON.stringify(x));
const fmt=d=>new Date(d+"T00:00:00").toLocaleDateString("en-GB",{day:"2-digit",month:"short",year:"numeric"});
const esc=s=>String(s||"").replace(/[&<>"']/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]));
function doLogin(){
 const u=document.getElementById("username").value.trim().toLowerCase(),p=document.getElementById("password").value.trim();
 const x=MANAGERS.find(a=>a.u===u&&a.p===p),err=document.getElementById("error");
 if(!x){err.textContent="Username or password is incorrect.";err.classList.remove("hidden");return}
 me=x;sessionStorage.setItem("10MU_USER",x.u);err.classList.add("hidden");document.getElementById("password").value="";openApp();
}
function openApp(){
 document.getElementById("login").style.display="none";document.getElementById("app").style.display="block";
 document.getElementById("userName").textContent=me.name;document.getElementById("userRole").textContent=me.role;document.getElementById("avatar").textContent=me.initial;
 document.getElementById("mineName").textContent=me.name;document.getElementById("mineRole").textContent=me.role;
 document.getElementById("towerDate").value=iso();document.getElementById("weekDate").value=iso();document.getElementById("monthDate").value=iso().slice(0,7);
 loadMine();renderTower();
}
function logout(){
  sessionStorage.removeItem("10MU_USER");
  me=null;
  document.getElementById("app").style.display="none";
  document.getElementById("login").style.display="grid";
  document.getElementById("username").value="";
  document.getElementById("password").value="";
  document.getElementById("error").classList.add("hidden");
  document.getElementById("username").focus();
}
function page(n,b){
 document.querySelectorAll(".nav").forEach(x=>x.classList.remove("active"));b.classList.add("active");
 ["tower","mine","weekly","monthly"].forEach(x=>document.getElementById(x).classList.toggle("hidden",x!==n));
 if(n==="tower")renderTower();if(n==="mine")loadMine();if(n==="weekly")renderWeekly();if(n==="monthly")renderMonthly();
}
function status(x,d){
 if(!x||!x.submitted)return ["Pending","gray"];
 return [new Date(x.submittedAt)<=new Date(d+"T16:00:00")?"On Time":"Late",new Date(x.submittedAt)<=new Date(d+"T16:00:00")?"green":"red"];
}
function pill(p){return '<span class="pill '+(p==="Escalation"?"red":p==="Attention"?"yellow":"blue")+'">'+esc(p)+'</span>'}
function loadMine(){
 const d=iso(),x=get(d,me.u)||{};document.getElementById("mineDate").textContent=fmt(d);
 document.getElementById("todayWork").value=x.todayWork||"";document.getElementById("issue").value=x.issue||"";document.getElementById("tomorrow").value=x.tomorrow||"";
 document.getElementById("priority").value=x.priority||"Normal";document.getElementById("support").value=x.support||"No support";renderMyStatus();
}
function submitUpdate(){
 const d=iso(),old=get(d,me.u)||{},w=document.getElementById("todayWork").value.trim(),t=document.getElementById("tomorrow").value.trim();
 if(!w||!t){alert("Please complete Today's Work and Tomorrow's Agenda.");return}
 const x={date:d,user:me.u,name:me.name,role:me.role,todayWork:w,issue:document.getElementById("issue").value.trim(),tomorrow:t,priority:document.getElementById("priority").value,support:document.getElementById("support").value,submitted:true,submittedAt:old.submittedAt||new Date().toISOString(),updatedAt:new Date().toISOString()};
 put(d,me.u,x);document.getElementById("saveMsg").textContent="✓ Saved "+new Date().toLocaleTimeString("en-GB",{hour:"2-digit",minute:"2-digit"});renderMyStatus();renderTower();
}
function renderMyStatus(){
 const x=get(iso(),me.u),box=document.getElementById("myStatus");
 if(!x){box.innerHTML="<b>Pending</b><br>No update submitted yet today.";return}
 const s=status(x,iso());box.innerHTML="<b>"+s[0]+"</b> · "+pill(x.priority)+"<br>Submitted: "+new Date(x.submittedAt).toLocaleString("en-GB")+"<br>Support: "+esc(x.support);
}
function renderTower(){
 const d=document.getElementById("towerDate").value||iso(),xs=MANAGERS.map(m=>get(d,m.u)),sub=xs.filter(x=>x?.submitted),att=sub.filter(x=>x.priority==="Attention"),escs=sub.filter(x=>x.priority==="Escalation");
 document.getElementById("towerSub").textContent="Daily manager updates for "+fmt(d);
 document.getElementById("km").textContent=11;document.getElementById("ks").textContent=sub.length;document.getElementById("kp").textContent=11-sub.length;document.getElementById("kc").textContent=Math.round(sub.length/11*100)+"%";document.getElementById("ka").textContent=att.length;document.getElementById("ke").textContent=escs.length;
 document.getElementById("brief").innerHTML="<b>"+sub.length+"/11</b> managers submitted today's update. "+(att.length+escs.length?("<b>"+(att.length+escs.length)+"</b> priority item(s) require management attention."):"No priority alerts have been raised.")+"<br>Control Tower is shared; each manager's editable area remains personal.";
 const alerts=[...escs,...att];document.getElementById("attention").innerHTML=alerts.length?alerts.map(x=>'<div class="att '+(x.priority==="Escalation"?"redline":"")+'"><b>'+esc(x.name)+'</b> '+pill(x.priority)+'<br>'+esc(x.issue||"Priority flagged without blocker detail.")+'</div>').join(""):"<div class='muted'>No management attention items.</div>";
 document.getElementById("managerRows").innerHTML=MANAGERS.map(m=>{const x=get(d,m.u),s=status(x,d);return "<tr><td><b>"+esc(m.name)+"</b><br><span class='muted'>"+esc(m.role)+"</span></td><td>"+(x?esc(x.todayWork):"—")+"</td><td>"+(x?esc(x.issue||"—"):"—")+"</td><td>"+(x?esc(x.tomorrow):"—")+"</td><td>"+(x?pill(x.priority):"—")+"</td><td><span class='pill "+s[1]+"'>"+s[0]+"</span></td><td>"+(x?new Date(x.submittedAt).toLocaleTimeString("en-GB",{hour:"2-digit",minute:"2-digit"}):"—")+"</td></tr>"}).join("");
}
function weekDates(a){
 let d=new Date(a+"T00:00:00"),day=d.getDay(),diff=day===0?-6:1-day;d.setDate(d.getDate()+diff);
 return Array.from({length:5},(_,i)=>{let z=new Date(d);z.setDate(d.getDate()+i);return z.toISOString().slice(0,10)});
}
function renderWeekly(){
 const ds=weekDates(document.getElementById("weekDate").value||iso());let total=0;
 MANAGERS.forEach(m=>ds.forEach(d=>{if(get(d,m.u)?.submitted)total++}));
 let h='<div class="grid three"><div class="card kpi"><div class="lab">SUBMISSIONS</div><div class="num">'+total+'</div></div><div class="card kpi"><div class="lab">COMPLIANCE</div><div class="num">'+Math.round(total/55*100)+'%</div></div><div class="card kpi"><div class="lab">PERIOD</div><div class="num" style="font-size:16px">'+fmt(ds[0])+'</div></div></div>';

 MANAGERS.forEach(m=>{let n=0,a=0,e=0;for(let i=1;i<=days;i++){let x=get(ym+"-"+String(i).padStart(2,"0"),m.u);if(x?.submitted){n++;if(x.priority==="Attention")a++;if(x.priority==="Escalation")e++}}h+="<tr><td><b>"+esc(m.name)+"</b></td><td>"+n+"</td><td>"+days+"</td><td><b>"+Math.round(n/days*100)+"%</b></td><td>"+a+"</td><td>"+e+"</td></tr>"});h+="</tbody></table></div></div>";document.getElementById("monthlyContent").innerHTML=h;
}
